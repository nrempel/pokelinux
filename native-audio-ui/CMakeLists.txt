cmake_minimum_required(VERSION 3.22)
project(pokelinux_audio_ui LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Network Quick QuickControls2 Test)
# One app-local Rust queue core; transport and provider authority stay outside it.
set(POKELINUX_SOUNDS_QUEUE_ARCHIVE "" CACHE FILEPATH "Explicit already-built queue core archive")
set(POKELINUX_SOUNDS_QUEUE_ARCHIVE_SHA256 "" CACHE STRING "Required digest for an explicit queue archive")
if(POKELINUX_SOUNDS_QUEUE_ARCHIVE)
 file(SHA256 "${POKELINUX_SOUNDS_QUEUE_ARCHIVE}" actual_queue_sha256)
 if(NOT POKELINUX_SOUNDS_QUEUE_ARCHIVE_SHA256 OR NOT actual_queue_sha256 STREQUAL POKELINUX_SOUNDS_QUEUE_ARCHIVE_SHA256)
  message(FATAL_ERROR "Explicit Sounds queue archive digest mismatch")
 endif()
 set(queue_archive "${POKELINUX_SOUNDS_QUEUE_ARCHIVE}")
else()
 if(CMAKE_CROSSCOMPILING)
  message(FATAL_ERROR "Cross builds require an explicit pinned Sounds queue archive for the matching target")
 endif()
 find_program(CARGO_EXECUTABLE cargo REQUIRED)
 set(queue_archive "${CMAKE_CURRENT_BINARY_DIR}/rust-target/release/libpokelinux_sounds_queue.a")
 # Cargo tracks all transitive path dependencies, including the shared clock.
 add_custom_target(sounds-queue-build
  COMMAND ${CMAKE_COMMAND} -E env "CARGO_TARGET_DIR=${CMAKE_CURRENT_BINARY_DIR}/rust-target"
   ${CARGO_EXECUTABLE} build --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/client-core/Cargo.toml" --locked --offline --release
  BYPRODUCTS "${queue_archive}"
  VERBATIM)
endif()
add_library(sounds-queue-core STATIC IMPORTED)
set_target_properties(sounds-queue-core PROPERTIES IMPORTED_LOCATION "${queue_archive}")
if(TARGET sounds-queue-build)
 add_dependencies(sounds-queue-core sounds-queue-build)
endif()
find_package(Threads REQUIRED)
add_library(audio-client STATIC AudioClient.cpp AudioClient.h)
target_include_directories(audio-client PUBLIC client-core/include)
target_link_libraries(audio-client PUBLIC Qt6::Core Qt6::Network sounds-queue-core Threads::Threads ${CMAKE_DL_LIBS} m)
target_compile_options(audio-client PRIVATE -Wall -Wextra -Werror)
add_library(phone-appearance-client STATIC ../native-app/PhoneAppearanceClient.cpp ../native-app/PhoneAppearanceClient.h)
target_include_directories(phone-appearance-client PUBLIC ../native-app)
target_link_libraries(phone-appearance-client PUBLIC Qt6::Core Qt6::Gui)
target_compile_options(phone-appearance-client PRIVATE -Wall -Wextra -Werror)
add_executable(pokelinux-sounds main.cpp resources.qrc)
target_include_directories(pokelinux-sounds PRIVATE ../native-app)
target_link_libraries(pokelinux-sounds PRIVATE audio-client phone-appearance-client Qt6::Quick Qt6::QuickControls2)
target_compile_options(pokelinux-sounds PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_executable(audio-client-test AudioClientTest.cpp)
target_link_libraries(audio-client-test PRIVATE audio-client Qt6::Test)
target_compile_options(audio-client-test PRIVATE -Wall -Wextra -Werror)
add_test(NAME audio-client-boundary COMMAND audio-client-test)
add_executable(sounds-queue-test client-core/tests/queue-functional.cpp)
target_include_directories(sounds-queue-test PRIVATE client-core/include)
target_link_libraries(sounds-queue-test PRIVATE sounds-queue-core Threads::Threads ${CMAKE_DL_LIBS} m)
target_compile_options(sounds-queue-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME sounds-queue-boundary COMMAND sounds-queue-test)
set_tests_properties(sounds-queue-boundary PROPERTIES TIMEOUT 8)
install(TARGETS pokelinux-sounds RUNTIME DESTINATION libexec/pokelinux)
install(FILES ../native-app/PhoneAppearance.qml DESTINATION share/pokelinux/appearance)
add_subdirectory(../native-app/appearance-client-tests appearance-client-tests)
install(FILES org.omarchy.Sounds.desktop DESTINATION share/applications)
