cmake_minimum_required(VERSION 3.22)
project(pokelinux_permissions_ui LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Quick QuickControls2)
find_package(Threads REQUIRED)
include("${CMAKE_CURRENT_LIST_DIR}/../native-app/AppearanceRustClient.cmake")
set(POKELINUX_PERMISSIONS_CLIENT_ARCHIVE "" CACHE FILEPATH "Explicit matching Permissions Rust client archive")
set(POKELINUX_PERMISSIONS_CLIENT_ARCHIVE_SHA256 "" CACHE STRING "Required SHA256 for explicit Permissions archive")
if((POKELINUX_PERMISSIONS_CLIENT_ARCHIVE AND NOT POKELINUX_PERMISSIONS_CLIENT_ARCHIVE_SHA256) OR (POKELINUX_PERMISSIONS_CLIENT_ARCHIVE_SHA256 AND NOT POKELINUX_PERMISSIONS_CLIENT_ARCHIVE))
 message(FATAL_ERROR "Explicit Permissions archive and SHA256 must be supplied together")
endif()
if(POKELINUX_PERMISSIONS_CLIENT_ARCHIVE)
 file(SHA256 "${POKELINUX_PERMISSIONS_CLIENT_ARCHIVE}" actual_permissions_client_sha256)
 if(NOT actual_permissions_client_sha256 STREQUAL POKELINUX_PERMISSIONS_CLIENT_ARCHIVE_SHA256)
  message(FATAL_ERROR "Explicit Permissions client archive digest mismatch")
 endif()
 set(permissions_client_archive "${POKELINUX_PERMISSIONS_CLIENT_ARCHIVE}")
else()
 if(CMAKE_CROSSCOMPILING)
  message(FATAL_ERROR "Cross builds require a pinned matching-target Permissions archive")
 endif()
 find_program(CARGO_EXECUTABLE cargo REQUIRED)
 set(permissions_client_archive "${CMAKE_CURRENT_BINARY_DIR}/rust-target/release/libpokelinux_permissions_client.a")
 add_custom_target(permissions-client-rust-build
  COMMAND ${CMAKE_COMMAND} -E env "CARGO_TARGET_DIR=${CMAKE_CURRENT_BINARY_DIR}/rust-target"
   ${CARGO_EXECUTABLE} build --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/client-core/Cargo.toml" --locked --offline --release
  BYPRODUCTS "${permissions_client_archive}" VERBATIM)
endif()
add_library(permissions-client-core STATIC IMPORTED)
set_target_properties(permissions-client-core PROPERTIES IMPORTED_LOCATION "${permissions_client_archive}"
 INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/client-core/include")
target_link_libraries(permissions-client-core INTERFACE Threads::Threads ${CMAKE_DL_LIBS} m)
if(TARGET permissions-client-rust-build)
 add_dependencies(permissions-client-core permissions-client-rust-build)
endif()
add_library(permissions-client STATIC PermissionsClient.cpp PermissionsClient.h)
target_include_directories(permissions-client PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(permissions-client PUBLIC Qt6::Core permissions-client-core)
target_compile_options(permissions-client PRIVATE -Wall -Wextra -Werror)
add_library(phone-appearance-client STATIC ../native-app/PhoneAppearanceClient.cpp ../native-app/PhoneAppearanceClient.h)
target_include_directories(phone-appearance-client PUBLIC ../native-app)
target_link_libraries(phone-appearance-client PUBLIC Qt6::Core Qt6::Gui appearance-client-core)
target_compile_options(phone-appearance-client PRIVATE -Wall -Wextra -Werror)
add_executable(pokelinux-permissions main.cpp resources.qrc)
target_link_libraries(pokelinux-permissions PRIVATE permissions-client phone-appearance-client Qt6::Quick Qt6::QuickControls2)
target_compile_options(pokelinux-permissions PRIVATE -Wall -Wextra -Werror)
find_package(Qt6 REQUIRED COMPONENTS Test)
add_executable(permissions-click-test tests/click.cpp resources.qrc)
target_link_libraries(permissions-click-test PRIVATE permissions-client phone-appearance-client Qt6::Quick Qt6::QuickControls2 Qt6::Test)
target_compile_options(permissions-click-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
# Protected Qt data symbols require PIC even in executables: -fPIE may copy them.
# Keep the C++ archive members PIC as well as the two final executables.
foreach(qt_target IN ITEMS permissions-client phone-appearance-client pokelinux-permissions permissions-click-test)
 target_compile_options(${qt_target} PRIVATE -fPIC)
endforeach()
include(CTest)
if(BUILD_TESTING)
 if(NOT CMAKE_READELF)
  find_program(CMAKE_READELF NAMES readelf llvm-readelf REQUIRED)
 endif()
 foreach(elf_target IN ITEMS pokelinux-permissions permissions-click-test)
  add_test(NAME ${elf_target}-no-copy-relocations
   COMMAND "${CMAKE_COMMAND}" "-DREADELF=${CMAKE_READELF}" "-DELF=$<TARGET_FILE:${elf_target}>"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/no-copy-relocations.cmake")
 endforeach()
endif()
install(TARGETS pokelinux-permissions RUNTIME DESTINATION libexec/pokelinux)
install(FILES ../native-app/PhoneAppearance.qml DESTINATION share/pokelinux/appearance)
install(FILES org.omarchy.Permissions.desktop DESTINATION share/applications)
