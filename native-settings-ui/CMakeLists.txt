cmake_minimum_required(VERSION 3.22)
project(pokelinux_settings_ui LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Quick QuickControls2)
find_package(Threads REQUIRED)
include("${CMAKE_CURRENT_LIST_DIR}/../native-app/AppearanceRustClient.cmake")
set(POKELINUX_SETTINGS_CLIENT_ARCHIVE "" CACHE FILEPATH "Explicit matching Settings Rust client archive")
set(POKELINUX_SETTINGS_CLIENT_ARCHIVE_SHA256 "" CACHE STRING "Required SHA256 for explicit Settings archive")
if((POKELINUX_SETTINGS_CLIENT_ARCHIVE AND NOT POKELINUX_SETTINGS_CLIENT_ARCHIVE_SHA256) OR (POKELINUX_SETTINGS_CLIENT_ARCHIVE_SHA256 AND NOT POKELINUX_SETTINGS_CLIENT_ARCHIVE))
 message(FATAL_ERROR "Explicit Settings archive and SHA256 must be supplied together")
endif()
if(POKELINUX_SETTINGS_CLIENT_ARCHIVE)
 file(SHA256 "${POKELINUX_SETTINGS_CLIENT_ARCHIVE}" actual_settings_client_sha256)
 if(NOT actual_settings_client_sha256 STREQUAL POKELINUX_SETTINGS_CLIENT_ARCHIVE_SHA256)
  message(FATAL_ERROR "Explicit Settings client archive digest mismatch")
 endif()
 set(settings_client_archive "${POKELINUX_SETTINGS_CLIENT_ARCHIVE}")
else()
 if(CMAKE_CROSSCOMPILING)
  message(FATAL_ERROR "Cross builds require a pinned matching-target Settings archive")
 endif()
 find_program(CARGO_EXECUTABLE cargo REQUIRED)
 set(settings_client_archive "${CMAKE_CURRENT_BINARY_DIR}/rust-target/release/libpokelinux_settings_client.a")
 add_custom_target(settings-client-rust-build
  COMMAND ${CMAKE_COMMAND} -E env "CARGO_TARGET_DIR=${CMAKE_CURRENT_BINARY_DIR}/rust-target"
   ${CARGO_EXECUTABLE} build --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/client-core/Cargo.toml" --locked --offline --release
  BYPRODUCTS "${settings_client_archive}" VERBATIM)
endif()
add_library(settings-client-core STATIC IMPORTED)
set_target_properties(settings-client-core PROPERTIES IMPORTED_LOCATION "${settings_client_archive}"
 INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/client-core/include")
target_link_libraries(settings-client-core INTERFACE Threads::Threads ${CMAKE_DL_LIBS} m)
if(TARGET settings-client-rust-build)
 add_dependencies(settings-client-core settings-client-rust-build)
endif()
add_library(settings-client STATIC SettingsClient.cpp SettingsClient.h)
target_include_directories(settings-client PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(settings-client PUBLIC Qt6::Core settings-client-core)
target_compile_options(settings-client PRIVATE -Wall -Wextra -Werror)
add_library(phone-appearance-client STATIC ../native-app/PhoneAppearanceClient.cpp ../native-app/PhoneAppearanceClient.h)
target_include_directories(phone-appearance-client PUBLIC ../native-app)
target_link_libraries(phone-appearance-client PUBLIC Qt6::Core Qt6::Gui appearance-client-core)
target_compile_options(phone-appearance-client PRIVATE -Wall -Wextra -Werror)
add_executable(pokelinux-settings main.cpp resources.qrc)
target_link_libraries(pokelinux-settings PRIVATE settings-client phone-appearance-client Qt6::Quick Qt6::QuickControls2)
target_compile_options(pokelinux-settings PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_executable(settings-client-component-test tests/component.cpp resources.qrc)
target_link_libraries(settings-client-component-test PRIVATE settings-client phone-appearance-client Qt6::Quick Qt6::QuickControls2)
target_compile_options(settings-client-component-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
# Run as UID11050 in an isolated root-authored environment with the installed QML.
# This component target does not create a normal app window or contact a service.
install(TARGETS pokelinux-settings RUNTIME DESTINATION libexec/pokelinux)
install(FILES ../native-app/PhoneAppearance.qml DESTINATION share/pokelinux/appearance)
install(FILES org.omarchy.Settings.desktop DESTINATION share/applications)
