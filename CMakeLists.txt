cmake_minimum_required(VERSION 3.22)
project(pixel_graphics_bridge C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(HYBRIS_SOURCE "" CACHE PATH "Pinned libhybris hybris/ directory")
if(NOT EXISTS "${HYBRIS_SOURCE}/egl/ws.h")
  message(FATAL_ERROR "Set HYBRIS_SOURCE to the pinned libhybris hybris/ directory")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(EGL REQUIRED IMPORTED_TARGET egl)
pkg_check_modules(GLES REQUIRED IMPORTED_TARGET glesv2)
pkg_check_modules(DRM REQUIRED IMPORTED_TARGET libdrm)
add_compile_options(-Wall -Wextra -Werror)
add_library(eglplatform_pixel MODULE src/eglplatform_pixel.c)
set_target_properties(eglplatform_pixel PROPERTIES PREFIX "")
target_include_directories(eglplatform_pixel PRIVATE "${HYBRIS_SOURCE}/egl")
target_link_libraries(eglplatform_pixel PRIVATE PkgConfig::EGL)
target_link_options(eglplatform_pixel PRIVATE -Wl,-z,defs)
add_executable(pixel-dmabuf-probe src/dmabuf_probe.c)
target_link_libraries(pixel-dmabuf-probe PRIVATE PkgConfig::EGL PkgConfig::GLES PkgConfig::DRM)
include(CTest)
if(BUILD_TESTING)
  add_executable(test-backend tests/test_backend.c src/eglplatform_pixel.c)
  target_include_directories(test-backend PRIVATE "${HYBRIS_SOURCE}/egl")
  target_link_libraries(test-backend PRIVATE PkgConfig::EGL)
  add_test(NAME backend-contracts COMMAND test-backend)
  add_executable(test-software-egl tests/test_software_egl.c)
  target_include_directories(test-software-egl PRIVATE "${HYBRIS_SOURCE}/egl")
  target_link_libraries(test-software-egl PRIVATE PkgConfig::EGL PkgConfig::GLES ${CMAKE_DL_LIBS})
  add_test(NAME software-egl COMMAND test-software-egl $<TARGET_FILE:eglplatform_pixel>)
  set_tests_properties(software-egl PROPERTIES ENVIRONMENT "LIBGL_ALWAYS_SOFTWARE=1" TIMEOUT 30)
  add_library(fake-android-egl MODULE tests/fake_android_egl.c)
  set_target_properties(fake-android-egl PROPERTIES PREFIX "")
  target_compile_options(fake-android-egl PRIVATE -fno-stack-protector)
  target_link_options(fake-android-egl PRIVATE -nostdlib)
  add_executable(test-hybris tests/test_hybris.c)
  target_link_libraries(test-hybris PRIVATE "${HYBRIS_SOURCE}/egl/.libs/libEGL.so")
  target_link_options(test-hybris PRIVATE "-Wl,-rpath-link,${HYBRIS_SOURCE}/common/.libs" "-Wl,-rpath-link,${HYBRIS_SOURCE}/hardware/.libs")
  add_test(NAME hybris-wrapper COMMAND test-hybris)
  set_tests_properties(hybris-wrapper PROPERTIES TIMEOUT 30 ENVIRONMENT
    "LD_LIBRARY_PATH=${HYBRIS_SOURCE}/common/.libs:${HYBRIS_SOURCE}/hardware/.libs:${HYBRIS_SOURCE}/egl/.libs;HYBRIS_ANDROID_SDK_VERSION=37;HYBRIS_LINKER_DIR=${HYBRIS_SOURCE}/common/q/.libs;HYBRIS_EGLPLATFORM=pixel;HYBRIS_EGLPLATFORM_DIR=$<TARGET_FILE_DIR:eglplatform_pixel>;LIBEGL=$<TARGET_FILE:fake-android-egl>;LIBGLESV2=$<TARGET_FILE:fake-android-egl>")
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME runtime-and-cli COMMAND ${Python3_EXECUTABLE} -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/tests" -p "test_*.py")
  set_tests_properties(runtime-and-cli PROPERTIES ENVIRONMENT "PIXEL_PROBE=$<TARGET_FILE:pixel-dmabuf-probe>")
endif()
install(TARGETS eglplatform_pixel LIBRARY DESTINATION lib/libhybris)
install(TARGETS pixel-dmabuf-probe RUNTIME DESTINATION bin)

add_executable(loader-probe src/loader_probe.c)
target_include_directories(loader-probe PRIVATE "${HYBRIS_SOURCE}/include")
target_link_libraries(loader-probe PRIVATE "${HYBRIS_SOURCE}/common/.libs/libhybris-common.so")
install(TARGETS loader-probe RUNTIME DESTINATION bin)

add_library(pixel-runtime SHARED src/pixel_runtime.c)
target_include_directories(pixel-runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src" PRIVATE "${HYBRIS_SOURCE}/include")
target_link_libraries(pixel-runtime PRIVATE "${HYBRIS_SOURCE}/common/.libs/libhybris-common.so")
add_executable(runtime-probe src/runtime_probe.c)
target_include_directories(runtime-probe PRIVATE "${HYBRIS_SOURCE}/include")
target_link_libraries(runtime-probe PRIVATE pixel-runtime "${HYBRIS_SOURCE}/common/.libs/libhybris-common.so" "${HYBRIS_SOURCE}/egl/.libs/libEGL.so")
if(BUILD_TESTING)
 add_executable(test-properties tests/test_properties.c)
 target_link_libraries(test-properties PRIVATE pixel-runtime pthread)
 add_test(NAME properties COMMAND test-properties)
endif()
install(TARGETS pixel-runtime LIBRARY DESTINATION lib)
install(TARGETS runtime-probe RUNTIME DESTINATION bin)
install(FILES src/pixel_runtime.h DESTINATION include)

target_link_libraries(pixel-dmabuf-probe PRIVATE pixel-runtime)

target_link_libraries(loader-probe PRIVATE pixel-runtime)

set(PIXEL_FACTORY_RUNTIME "" CACHE PATH "Optional verified factory payload for real Android ABI tests")
if(BUILD_TESTING AND PIXEL_FACTORY_RUNTIME)
 find_program(QEMU_AARCH64 qemu-aarch64 REQUIRED)
 add_library(stock-android-fixture MODULE tests/stock_android_fixture.c)
 set_target_properties(stock-android-fixture PROPERTIES PREFIX "")
 target_compile_options(stock-android-fixture PRIVATE -fno-builtin -fno-stack-protector)
 target_link_options(stock-android-fixture PRIVATE -nostdlib -Wl,--no-as-needed)
 target_link_libraries(stock-android-fixture PRIVATE
   "${PIXEL_FACTORY_RUNTIME}/apex/com.android.runtime/lib64/bionic/libc.so"
   "${PIXEL_FACTORY_RUNTIME}/apex/com.android.runtime/lib64/bionic/libm.so")
 add_test(NAME stock-android-abi COMMAND ${QEMU_AARCH64} -cpu max -L "${PIXEL_FACTORY_RUNTIME}"
   $<TARGET_FILE:runtime-probe> "${PIXEL_FACTORY_RUNTIME}" --smoke $<TARGET_FILE:stock-android-fixture>)
 foreach(driver libEGL_powervr.so libGLESv2_powervr.so)
   add_test(NAME stock-load-${driver} COMMAND ${QEMU_AARCH64} -cpu max -L "${PIXEL_FACTORY_RUNTIME}"
     $<TARGET_FILE:runtime-probe> "${PIXEL_FACTORY_RUNTIME}" --load "${PIXEL_FACTORY_RUNTIME}/vendor/lib64/egl/${driver}")
 endforeach()
 set_tests_properties(stock-android-abi stock-load-libEGL_powervr.so stock-load-libGLESv2_powervr.so
   PROPERTIES TIMEOUT 60 ENVIRONMENT
   "HYBRIS_ANDROID_SDK_VERSION=37;HYBRIS_LINKER_DIR=${HYBRIS_SOURCE}/common/q/.libs;HYBRIS_LD_LIBRARY_PATH=${PIXEL_FACTORY_RUNTIME}/vendor/lib64:${PIXEL_FACTORY_RUNTIME}/system/lib64:${PIXEL_FACTORY_RUNTIME}/apex/com.android.runtime/lib64/bionic:${PIXEL_FACTORY_RUNTIME}/apex/com.android.runtime/lib64")
 set_property(TEST runtime-and-cli APPEND PROPERTY ENVIRONMENT "PIXEL_RUNTIME=${PIXEL_FACTORY_RUNTIME}")
endif()

if(BUILD_TESTING AND CMAKE_CROSSCOMPILING)
 string(REPLACE ";" " " PIXEL_EMULATOR_COMMAND "${CMAKE_CROSSCOMPILING_EMULATOR}")
 set_property(TEST runtime-and-cli APPEND PROPERTY ENVIRONMENT "PIXEL_PROBE_EMULATOR=${PIXEL_EMULATOR_COMMAND}")
endif()
