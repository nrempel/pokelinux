cmake_minimum_required(VERSION 3.21)
project(NativeThemeWorkerBoundary LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
add_executable(worker-probe process.cpp "${THEME_CONSUMER_DIR}/PhoneThemeFiles.cpp" "${THEME_CONSUMER_DIR}/PhoneThemeFiles.h")
target_include_directories(worker-probe PRIVATE "${THEME_CONSUMER_DIR}")
target_link_libraries(worker-probe PRIVATE Qt6::Core)
target_compile_options(worker-probe PRIVATE -Wall -Wextra -Werror)
target_link_options(worker-probe PRIVATE -Wl,--wrap=_ZN8QProcess15waitForFinishedEi -Wl,--wrap=clock_gettime)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/qt-version.txt" "${Qt6Core_VERSION}\n")
