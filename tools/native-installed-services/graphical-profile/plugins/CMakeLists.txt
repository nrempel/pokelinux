project(pokelinux_selected_native_plugins LANGUAGES CXX)

 message(FATAL_ERROR "Selected native AArch64 build required")
endif()

set(CMAKE_CXX_EXTENSIONS OFF)

file(SHA256 "/usr/bin/Hyprland" compositor_hash)

 message(FATAL_ERROR "Exact selected Hyprland required")

if(NOT selected_hash STREQUAL "7d78b090b4fac0701fa5926d5a5bb53789d01b9309296e4fa8103e0b7f5643a4")
 message(FATAL_ERROR "Selected source-adapter.cpp bytes changed")
endif()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../../../native-session-compositor/source_sampler.hpp" selected_hash)
if(NOT selected_hash STREQUAL "73612325a597b71a4386d62a4a563477e78e1bd5e572e48784f005507600e845")
 message(FATAL_ERROR "Selected source_sampler.hpp bytes changed")
endif()
