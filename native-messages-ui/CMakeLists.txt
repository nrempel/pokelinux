cmake_minimum_required(VERSION 3.22)
project(pokelinux_messages_ui LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Quick QuickControls2)
include("${CMAKE_CURRENT_LIST_DIR}/../native-app/AppearanceRustClient.cmake")
option(MESSAGES_ISOLATED_OBSERVER "Build only disposable renderer qualification observer" OFF)
# Select one full Messages Rust engine; cross builds supply the matching archive.
set(POKELINUX_MESSAGES_CLIENT_ARCHIVE "" CACHE FILEPATH "Explicit already-built Messages client engine archive")
set(POKELINUX_MESSAGES_CLIENT_ARCHIVE_SHA256 "" CACHE STRING "Required digest for an explicit Messages client engine archive")
if((POKELINUX_MESSAGES_CLIENT_ARCHIVE AND NOT POKELINUX_MESSAGES_CLIENT_ARCHIVE_SHA256) OR (POKELINUX_MESSAGES_CLIENT_ARCHIVE_SHA256 AND NOT POKELINUX_MESSAGES_CLIENT_ARCHIVE))
 message(FATAL_ERROR "Explicit Messages archive and SHA256 must be supplied together")
endif()
if(POKELINUX_MESSAGES_CLIENT_ARCHIVE)
 file(SHA256 "${POKELINUX_MESSAGES_CLIENT_ARCHIVE}" actual_messages_client_sha256)
 if(NOT actual_messages_client_sha256 STREQUAL POKELINUX_MESSAGES_CLIENT_ARCHIVE_SHA256)
  message(FATAL_ERROR "Explicit Messages client archive digest mismatch")
 endif()
 set(messages_client_archive "${POKELINUX_MESSAGES_CLIENT_ARCHIVE}")
else()
 if(CMAKE_CROSSCOMPILING)
  message(FATAL_ERROR "Cross builds require an explicit pinned Messages client archive for the matching target")
 endif()
 find_program(CARGO_EXECUTABLE cargo REQUIRED)
 set(messages_client_archive "${CMAKE_CURRENT_BINARY_DIR}/rust-target/release/libpokelinux_messages_client.a")
 # Always invoke Cargo: it tracks transitive path dependencies, including clocks.
 add_custom_target(messages-client-rust-build
  COMMAND ${CMAKE_COMMAND} -E env "CARGO_TARGET_DIR=${CMAKE_CURRENT_BINARY_DIR}/rust-target"
   ${CARGO_EXECUTABLE} build --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/client-core/Cargo.toml" --locked --offline --release
  BYPRODUCTS "${messages_client_archive}"
  VERBATIM)
endif()
add_library(messages-client-core STATIC IMPORTED)
set_target_properties(messages-client-core PROPERTIES
 IMPORTED_LOCATION "${messages_client_archive}"
 INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/client-core/include")
if(TARGET messages-client-rust-build)
 add_dependencies(messages-client-core messages-client-rust-build)
endif()
find_package(Threads REQUIRED)
add_library(messages-client STATIC MessagesClient.cpp MessagesClient.h)
target_include_directories(messages-client PUBLIC client-core/include)
target_link_libraries(messages-client PUBLIC Qt6::Core messages-client-core Threads::Threads ${CMAKE_DL_LIBS} m)
target_compile_options(messages-client PRIVATE -Wall -Wextra -Werror)
add_library(phone-appearance-client STATIC ../native-app/PhoneAppearanceClient.cpp ../native-app/PhoneAppearanceClient.h)
target_include_directories(phone-appearance-client PUBLIC ../native-app)
target_link_libraries(phone-appearance-client PUBLIC Qt6::Core Qt6::Gui appearance-client-core)
target_compile_options(phone-appearance-client PRIVATE -Wall -Wextra -Werror)
add_executable(pokelinux-messages main.cpp resources.qrc)
target_link_libraries(pokelinux-messages PRIVATE messages-client phone-appearance-client Qt6::Quick Qt6::QuickControls2)
target_compile_options(pokelinux-messages PRIVATE -Wall -Wextra -Werror)
if(MESSAGES_ISOLATED_OBSERVER)
 target_compile_definitions(pokelinux-messages PRIVATE MESSAGES_ISOLATED_OBSERVER)
endif()
enable_testing()
add_executable(messages-client-test client-core/tests/client-functional.cpp)
target_link_libraries(messages-client-test PRIVATE messages-client-core Threads::Threads ${CMAKE_DL_LIBS} m)
target_compile_options(messages-client-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME messages-client-boundary COMMAND messages-client-test)
set_tests_properties(messages-client-boundary PROPERTIES TIMEOUT 20)
# Build the unchanged shared appearance probe; its exact29 selected root/asset
# invocations remain with the owned boundary runner, as in Sounds.
add_subdirectory(../native-app/appearance-client-tests appearance-client-tests)
install(TARGETS pokelinux-messages RUNTIME DESTINATION libexec/pokelinux)
install(FILES ../native-app/PhoneAppearance.qml DESTINATION share/pokelinux/appearance)
install(FILES org.omarchy.Messages.desktop DESTINATION share/applications)
