cmake_minimum_required(VERSION 3.22)
project(pokelinux_messages_ui LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Quick QuickControls2)
option(MESSAGES_ISOLATED_OBSERVER "Build only disposable renderer qualification observer" OFF)
add_executable(pokelinux-messages main.cpp MessagesClient.cpp MessagesClient.h ../native-app/PhoneThemeFiles.cpp ../native-app/PhoneThemeFiles.h resources.qrc)
target_include_directories(pokelinux-messages PRIVATE ../native-app)
target_link_libraries(pokelinux-messages PRIVATE Qt6::Core Qt6::Quick Qt6::QuickControls2)
target_compile_options(pokelinux-messages PRIVATE -Wall -Wextra -Werror)
if(MESSAGES_ISOLATED_OBSERVER)
 target_compile_definitions(pokelinux-messages PRIVATE MESSAGES_ISOLATED_OBSERVER)
endif()
install(TARGETS pokelinux-messages RUNTIME DESTINATION libexec/pokelinux)
install(FILES ../native-app/PhoneTheme.qml ../native-app/ThemePalette.js DESTINATION share/pokelinux/appearance)

install(FILES org.omarchy.Messages.desktop DESTINATION share/applications)
