cmake_minimum_required(VERSION 3.22)
project(NativeMessagesWorkerBoundary LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
set(MESSAGES_CONSUMER_DIR "${CMAKE_CURRENT_LIST_DIR}/.." CACHE PATH "Exact current MessagesClient source and copied engine header directory")
# Select one full Messages Rust engine; cross builds supply the matching archive.
set(POKELINUX_MESSAGES_CLIENT_ARCHIVE "" CACHE FILEPATH "Explicit already-built Messages client engine archive")
set(POKELINUX_MESSAGES_CLIENT_ARCHIVE_SHA256 "" CACHE STRING "Required digest for an explicit Messages client engine archive")
if((POKELINUX_MESSAGES_CLIENT_ARCHIVE AND NOT POKELINUX_MESSAGES_CLIENT_ARCHIVE_SHA256) OR (POKELINUX_MESSAGES_CLIENT_ARCHIVE_SHA256 AND NOT POKELINUX_MESSAGES_CLIENT_ARCHIVE))
 message(FATAL_ERROR "Explicit Messages archive and SHA256 must be supplied together")
endif()
if(POKELINUX_MESSAGES_CLIENT_ARCHIVE)
 file(SHA256 "${POKELINUX_MESSAGES_CLIENT_ARCHIVE}" actual_messages_client_sha256)
 if(NOT actual_messages_client_sha256 STREQUAL POKELINUX_MESSAGES_CLIENT_ARCHIVE_SHA256)
  message(FATAL_ERROR "Explicit Messages client archive digest mismatch")
 endif()
 set(messages_client_archive "${POKELINUX_MESSAGES_CLIENT_ARCHIVE}")
else()
 if(CMAKE_CROSSCOMPILING)
  message(FATAL_ERROR "Cross builds require an explicit pinned Messages client archive for the matching target")
 endif()
 find_program(CARGO_EXECUTABLE cargo REQUIRED)
 set(messages_client_archive "${CMAKE_CURRENT_BINARY_DIR}/rust-target/release/libpokelinux_messages_client.a")
 # Always invoke Cargo: it tracks transitive path dependencies, including clocks.
 add_custom_target(messages-client-rust-build
  COMMAND ${CMAKE_COMMAND} -E env "CARGO_TARGET_DIR=${CMAKE_CURRENT_BINARY_DIR}/rust-target"
   ${CARGO_EXECUTABLE} build --manifest-path "${CMAKE_CURRENT_LIST_DIR}/../client-core/Cargo.toml" --locked --offline --release
  BYPRODUCTS "${messages_client_archive}"
  VERBATIM)
endif()
add_library(messages-client-core STATIC IMPORTED)
set_target_properties(messages-client-core PROPERTIES
 IMPORTED_LOCATION "${messages_client_archive}"
 INTERFACE_INCLUDE_DIRECTORIES "${MESSAGES_CONSUMER_DIR}/client-core/include")
if(TARGET messages-client-rust-build)
 add_dependencies(messages-client-core messages-client-rust-build)
endif()
find_package(Threads REQUIRED)
add_executable(messages-worker-probe process.cpp "${MESSAGES_CONSUMER_DIR}/MessagesClient.cpp" "${MESSAGES_CONSUMER_DIR}/MessagesClient.h")
target_include_directories(messages-worker-probe PRIVATE "${MESSAGES_CONSUMER_DIR}")
target_link_libraries(messages-worker-probe PRIVATE Qt6::Core messages-client-core Threads::Threads ${CMAKE_DL_LIBS} m)
target_compile_options(messages-worker-probe PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_options(messages-worker-probe PRIVATE -Wl,--wrap=_ZN7QThread4waitE14QDeadlineTimer -Wl,--wrap=clock_gettime -Wl,--wrap=recvmsg)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/qt-version.txt" "${Qt6Core_VERSION}\n")

# Permanent independent regressions use the current adapter/header and the same
# selected engine. Private baseline snapshots and their hash gates stay private.
add_executable(messages-client-qt-order-test qt-order.cpp "${MESSAGES_CONSUMER_DIR}/MessagesClient.cpp" "${MESSAGES_CONSUMER_DIR}/MessagesClient.h")
target_include_directories(messages-client-qt-order-test PRIVATE "${MESSAGES_CONSUMER_DIR}")
target_link_libraries(messages-client-qt-order-test PRIVATE Qt6::Core messages-client-core Threads::Threads ${CMAKE_DL_LIBS} m)
target_compile_options(messages-client-qt-order-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
foreach(symbol IN ITEMS connect send recvmsg close pr_messages_client_step_v1 pr_messages_client_received_v1 pr_messages_client_report_v1 pr_messages_client_deliver_v1 pr_messages_client_stop_v1 pr_messages_client_worker_exited_v1 pr_messages_client_join_result_v1 pr_messages_client_free_v1 _ZN7QThread4waitE14QDeadlineTimer)
 target_link_options(messages-client-qt-order-test PRIVATE "-Wl,--wrap=${symbol}")
endforeach()
add_executable(messages-pre-entry-join-test pre-entry-join.cpp "${MESSAGES_CONSUMER_DIR}/MessagesClient.cpp" "${MESSAGES_CONSUMER_DIR}/MessagesClient.h")
target_include_directories(messages-pre-entry-join-test PRIVATE "${MESSAGES_CONSUMER_DIR}")
target_link_libraries(messages-pre-entry-join-test PRIVATE Qt6::Core messages-client-core Threads::Threads ${CMAKE_DL_LIBS} m)
target_compile_options(messages-pre-entry-join-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
foreach(symbol IN ITEMS pr_messages_client_worker_enter_v1 pr_messages_client_stop_v1 _ZN7QThread4waitE14QDeadlineTimer pr_messages_client_join_result_v1 pr_messages_client_step_v1 pr_messages_client_worker_exited_v1 pr_messages_client_free_v1 socket)
 target_link_options(messages-pre-entry-join-test PRIVATE "-Wl,--wrap=${symbol}")
endforeach()
enable_testing()
foreach(case IN ITEMS notify-deliver-before-ack stale-deliver-before-ack stop-connect-entered stop-send-entered stop-notify-unacked)
 add_test(NAME "messages-qt-order-${case}" COMMAND messages-client-qt-order-test "${case}")
 set_tests_properties("messages-qt-order-${case}" PROPERTIES TIMEOUT 20)
endforeach()
add_test(NAME messages-pre-entry-join COMMAND messages-pre-entry-join-test)
set_tests_properties(messages-pre-entry-join PROPERTIES TIMEOUT 8)

# Whole-client numeric-domain regression, including actual Qt projection.
add_executable(messages-overflow-projection-test overflow-projection.cpp "${MESSAGES_CONSUMER_DIR}/MessagesClient.cpp" "${MESSAGES_CONSUMER_DIR}/MessagesClient.h")
target_include_directories(messages-overflow-projection-test PRIVATE "${MESSAGES_CONSUMER_DIR}")
target_link_libraries(messages-overflow-projection-test PRIVATE Qt6::Core messages-client-core Threads::Threads ${CMAKE_DL_LIBS} m)
target_compile_options(messages-overflow-projection-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME messages-overflow-projection COMMAND messages-overflow-projection-test)
set_tests_properties(messages-overflow-projection PROPERTIES TIMEOUT 8)

# Permanent copies retain the original 37 parser assertions, three recorded
# observations, navigation/generation/queued-delivery and denial controls.
add_executable(messages-current-qt-test current-qt-behavior.cpp "${MESSAGES_CONSUMER_DIR}/MessagesClient.cpp" "${MESSAGES_CONSUMER_DIR}/MessagesClient.h")
target_include_directories(messages-current-qt-test PRIVATE "${MESSAGES_CONSUMER_DIR}")
target_link_libraries(messages-current-qt-test PRIVATE Qt6::Core messages-client-core Threads::Threads ${CMAKE_DL_LIBS} m)
target_compile_options(messages-current-qt-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
foreach(case IN ITEMS parser navigation inflight-stale queued-stale delayed-ui denial)
 add_test(NAME "messages-current-${case}" COMMAND messages-current-qt-test "${case}")
 if(case STREQUAL "parser")
  set_tests_properties("messages-current-${case}" PROPERTIES TIMEOUT 20)
 else()
  set_tests_properties("messages-current-${case}" PROPERTIES TIMEOUT 8)
 endif()
endforeach()
