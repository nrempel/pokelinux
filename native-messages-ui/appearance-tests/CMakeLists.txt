cmake_minimum_required(VERSION 3.22)
project(MessagesInstalledAppearanceBoundary LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
add_executable(messages-appearance-boundary process.cpp ../MessagesClient.cpp ../MessagesClient.h ../../native-app/PhoneAppearanceClient.cpp ../../native-app/PhoneAppearanceClient.h ../resources.qrc)
target_include_directories(messages-appearance-boundary PRIVATE .. ../../native-app)
target_link_libraries(messages-appearance-boundary PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2)
target_compile_options(messages-appearance-boundary PRIVATE -Wall -Wextra -Werror)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/qt-version.txt" "${Qt6Core_VERSION}\n")
