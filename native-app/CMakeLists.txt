cmake_minimum_required(VERSION 3.21)
project(OmarchyPhoneNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
configure_file(org.omarchy.Phone.service.in org.omarchy.Phone.service @ONLY)
find_package(Qt6 REQUIRED COMPONENTS Core Test Qml DBus Quick QuickControls2)
add_library(phone-rpc STATIC PhoneRpc.cpp PhoneRpc.h PhonePower.cpp PhonePower.h PhoneThemeFiles.cpp PhoneThemeFiles.h)
target_link_libraries(phone-rpc PUBLIC Qt6::Core Qt6::DBus)
target_include_directories(phone-rpc PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(phone-rpc PRIVATE -Wall -Wextra -Werror)
add_executable(omarchy-phone-app main.cpp)
target_link_libraries(omarchy-phone-app PRIVATE phone-rpc Qt6::Qml Qt6::Quick Qt6::QuickControls2)
target_compile_options(omarchy-phone-app PRIVATE -Wall -Wextra -Werror)
qt_add_resources(omarchy-phone-app phone-ui PREFIX "/phone" FILES
    Main.qml PhoneView.qml IntentButton.qml PhoneSession.qml PhoneActions.qml
    PhoneLifecycle.qml PhoneObservations.qml PhoneTheme.qml ThemePalette.js)
install(TARGETS omarchy-phone-app RUNTIME DESTINATION bin)
install(FILES org.omarchy.Phone.desktop DESTINATION share/applications)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/org.omarchy.Phone.service" DESTINATION share/dbus-1/services)
include(CTest)
add_executable(test-phone-rpc tests/test_rpc.cpp)
target_link_libraries(test-phone-rpc PRIVATE phone-rpc Qt6::Test Qt6::Qml)
add_test(NAME phone-rpc COMMAND test-phone-rpc)
set_tests_properties(phone-rpc PROPERTIES TIMEOUT 20)

target_compile_definitions(test-phone-rpc PRIVATE PHONE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")

add_executable(test-phone-actions tests/test_actions.cpp)
target_link_libraries(test-phone-actions PRIVATE phone-rpc Qt6::Test Qt6::Qml)
target_compile_definitions(test-phone-actions PRIVATE PHONE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME phone-actions COMMAND test-phone-actions)
set_tests_properties(phone-actions PROPERTIES TIMEOUT 20)

add_executable(test-phone-power tests/test_power.cpp)
target_link_libraries(test-phone-power PRIVATE phone-rpc Qt6::Test Qt6::DBus Qt6::Qml)
target_compile_definitions(test-phone-power PRIVATE PHONE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME phone-power COMMAND test-phone-power)
set_tests_properties(phone-power PROPERTIES TIMEOUT 20)

add_executable(test-phone-view tests/test_view.cpp)
target_link_libraries(test-phone-view PRIVATE phone-rpc Qt6::Test Qt6::Qml Qt6::Quick Qt6::QuickControls2)
target_compile_definitions(test-phone-view PRIVATE PHONE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME phone-view COMMAND test-phone-view)
set_tests_properties(phone-view PROPERTIES TIMEOUT 20 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_QUICK_CONTROLS_STYLE=Basic")
