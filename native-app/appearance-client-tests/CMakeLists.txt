cmake_minimum_required(VERSION 3.22)
project(NativeAppearanceSocketBoundary LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Qml Quick)
add_executable(appearance-socket-probe process.cpp ../PhoneAppearanceClient.cpp ../PhoneAppearanceClient.h)
target_include_directories(appearance-socket-probe PRIVATE ..)
target_link_libraries(appearance-socket-probe PRIVATE Qt6::Core Qt6::Qml Qt6::Quick)
target_link_options(appearance-socket-probe PRIVATE -Wl,--wrap=recvmsg)
target_compile_options(appearance-socket-probe PRIVATE -Wall -Wextra -Werror)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/qt-version.txt" "${Qt6Core_VERSION}\n")
