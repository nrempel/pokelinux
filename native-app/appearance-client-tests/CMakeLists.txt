cmake_minimum_required(VERSION 3.22)
project(NativeAppearanceSocketBoundary LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Qml Quick)
include("${CMAKE_CURRENT_LIST_DIR}/../AppearanceRustClient.cmake")
add_executable(appearance-socket-probe process.cpp ../PhoneAppearanceClient.cpp ../PhoneAppearanceClient.h)
target_include_directories(appearance-socket-probe PRIVATE ..)
target_link_libraries(appearance-socket-probe PRIVATE Qt6::Core Qt6::Qml Qt6::Quick appearance-client-core)
target_link_options(appearance-socket-probe PRIVATE -Wl,--wrap=recvmsg)
target_compile_options(appearance-socket-probe PRIVATE -Wall -Wextra -Werror)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/qt-version.txt" "${Qt6Core_VERSION}\n")

enable_testing()
add_executable(appearance-client-test ../appearance-client-core/tests/client-functional.cpp)
target_link_libraries(appearance-client-test PRIVATE appearance-client-core)
target_link_options(appearance-client-test PRIVATE -Wl,--wrap=clock_gettime)
target_compile_options(appearance-client-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME appearance-client-boundary COMMAND appearance-client-test)
set_tests_properties(appearance-client-boundary PROPERTIES TIMEOUT 8)
